// ====== CONFIGURACIÓN ====== var TOKEN = 'sdc-habitos-2026-x7f2'; var HOJA_HABITOS = 'Habitos'; var HOJA_CONFIG = 'Habitos_Config'; function doGet(e) { var accion = e.parameter.action; if (accion === 'data') { return responderJSON(obtenerDatos_()); } // Cualquier otra llamada es de escritura: exige token if (e.parameter.token !== TOKEN) { return responderJSON({ ok: false, error: 'Token inválido' }); } if (e.parameter.hoja === 'habito') { return responderJSON(marcarHabito_(e.parameter.fecha, e.parameter.habito, e.parameter.cumplido === 'true')); } if (e.parameter.hoja === 'habito_config') { if (e.parameter.accion === 'agregar') { return responderJSON(agregarHabito_(e.parameter.nombre, e.parameter.categoria)); } if (e.parameter.accion === 'archivar') { return responderJSON(archivarHabito_(e.parameter.nombre)); } } return responderJSON({ ok: false, error: 'Acción no reconocida' }); } function agregarHabito_(nombre, categoria) { nombre = (nombre || '').trim(); if (!nombre) return { ok: false, error: 'Falta el nombre del hábito' }; var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CONFIG); var datos = hoja.getDataRange().getValues(); var headers = datos[0]; var colHabito = headers.indexOf('Habito'); var colOrden = headers.indexOf('Orden'); var yaExiste = datos.slice(1).some(function (f) { return f[colHabito] === nombre; }); if (yaExiste) return { ok: false, error: 'Ese hábito ya existe' }; var maxOrden = datos.slice(1).reduce(function (max, f) { var o = Number(f[colOrden]) || 0; return o > max ? o : max; }, 0); hoja.appendRow([nombre, (categoria || '').trim(), true, maxOrden + 1]); return { ok: true }; } function archivarHabito_(nombre) { nombre = (nombre || '').trim(); if (!nombre) return { ok: false, error: 'Falta el nombre del hábito' }; var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CONFIG); var datos = hoja.getDataRange().getValues(); var headers = datos[0]; var colHabito = headers.indexOf('Habito'); var colActivo = headers.indexOf('Activo'); for (var i = 1; i < datos.length; i++) { if (datos[i][colHabito] === nombre) { hoja.getRange(i + 1, colActivo + 1).setValue(false); return { ok: true }; } } return { ok: false, error: 'No se encontró ese hábito' }; } function obtenerDatos_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var config = hojaAObjetos_(ss.getSheetByName(HOJA_CONFIG)) .filter(function (h) { return h.Activo === true || h.Activo === 'TRUE'; }) .sort(function (a, b) { return (a.Orden || 0) - (b.Orden || 0); }); var registros = hojaAObjetos_(ss.getSheetByName(HOJA_HABITOS)); return { ok: true, config: config, registros: registros }; } function marcarHabito_(fecha, habito, cumplido) { if (!fecha || !habito) return { ok: false, error: 'Falta fecha o hábito' }; var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_HABITOS); var datos = hoja.getDataRange().getValues(); var headers = datos[0]; var colFecha = headers.indexOf('Fecha'); var colHabito = headers.indexOf('Habito'); var colCumplido = headers.indexOf('Cumplido'); for (var i = 1; i < datos.length; i++) { var filaFecha = formatearFecha_(datos[i][colFecha]); if (filaFecha === fecha && datos[i][colHabito] === habito) { hoja.getRange(i + 1, colCumplido + 1).setValue(cumplido); return { ok: true, actualizado: true }; } } hoja.appendRow([fecha, habito, cumplido, '']); return { ok: true, actualizado: false, fila_insertada: hoja.getLastRow() }; } function hojaAObjetos_(hoja) { if (!hoja) return []; var datos = hoja.getDataRange().getValues(); var headers = datos.shift(); return datos .filter(function (fila) { return fila.some(function (v) { return v !== '' && v !== null; }); }) .map(function (fila) { var obj = {}; headers.forEach(function (h, i) { var valor = fila[i]; if (h === 'Fecha' && valor instanceof Date) valor = formatearFecha_(valor); obj[h] = valor; }); return obj; }); } function formatearFecha_(valor) { if (valor instanceof Date) { return Utilities.formatDate(valor, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return String(valor); } function responderJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
